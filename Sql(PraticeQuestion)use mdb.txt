Q31. Find the difference between the highest and lowest salary in each department.

SELECT e.dep_id,
       d.dept_name,
       MAX(e.salary) AS highest_salary,
       MIN(e.salary) AS lowest_salary,
       MAX(e.salary) - MIN(salary) AS salary_difference
FROM employees e
join departments d on e.dep_id = d.dep_id 
GROUP BY e.dep_id;
